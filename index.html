function collideBallBlock(ball,b){ if(b.dead)return false; const cooldown = ball.hitCooldown.get(b) || 0; if(cooldown>0)return false; const closestX = clamp(ball.x,b.x,b.x+b.w); const closestY = clamp(ball.y,b.y,b.y+b.h); let dx=ball.x-closestX; let dy=ball.y-closestY; const distSq=dx*dx+dy*dy; if(distSq>ball.r*ball.r)return false; let nx,ny,penetration; if(distSq===0){ const left=Math.abs(ball.x-b.x); const right=Math.abs(ball.x-(b.x+b.w)); const top=Math.abs(ball.y-b.y); const bottom=Math.abs(ball.y-(b.y+b.h)); const min=Math.min(left,right,top,bottom); if(min===left){ nx=-1; ny=0; penetration=ball.r+left; } else if(min===right){ nx=1; ny=0; penetration=ball.r+right; } else if(min===top){ nx=0; ny=-1; penetration=ball.r+top; } else{ nx=0; ny=1; penetration=ball.r+bottom; } }else{ const dist=Math.sqrt(distSq); nx=dx/dist; ny=dy/dist; penetration=ball.r-dist; } /* Push ball away from block */ ball.x+=nx*(penetration+1); ball.y+=ny*(penetration+1); /* Bounce */ const velocityIntoBlock= ball.vx*nx+ ball.vy*ny; if(velocityIntoBlock<0){ ball.vx-=2*velocityIntoBlock*nx; ball.vy-=2*velocityIntoBlock*ny; } /* Collision cooldown */ ball.hitCooldown.set(b,0.08); /* Damage */ const damage=ballDamage(ball.type); b.hp-=damage; b.flash=1; explode(ball.x,ball.y,false); shake=Math.min( 8, shake+.45 ); /* Block destroyed */ if(b.hp<=0){ b.hp=0; b.dead=true; const reward= Math.floor( 5+ data.level*2+ b.maxHp*.25 ); data.credits+=reward; floatText( b.x+b.w/2, b.y+b.h/2, "+"+format(reward) ); explode( b.x+b.w/2, b.y+b.h/2, true ); shake=Math.min( 14, shake+3 ); } return true; }
